T6の変更内容を master ブランチで確認しました。`scripts/ingest_events.py` には `--limit` オプションが追加され、`argparse` で引数を受け取り `ingest()` に渡すようになっています。また `_parse_ts` で年0や無効な ISO 形式を `None` にフォールバックする実装が含まれており、参加者の役割が欠落した際には `role` を `"participant"` に補うロジックが残っています。イベント参加者テーブルの `role` 列にデフォルト値を設定するマイグレーション (`db/migrations/2025-08-27_event_participant_role_default.sql`) も存在し、NOT NULL 制約違反を避けられるようになっています。

T6指示書では、こうした変更点を踏まえて環境変数の設定、スクリプト単体試験、タイムスタンプガードと役割デフォルトの確認、systemd サービスのログ監視、そしてメトリクスの増加確認といったDoDが整理されていました。ユーザーからの報告によると、これらの検証を実行して `ingested_events=N` を確認し、エラーが出ないことを確認済みとのことです。

# ① 収集スクリプト（CSV追記）

```bash
# 1) 保存先ディレクトリ

# 2) 収集スクリプト配置

set -euo pipefail

# 設定読込（DATABASE_URL 等）
if [[ -f /etc/default/mcp-news ]]; then
  set -a; source /etc/default/mcp-news; set +a
fi

CSV="/var/log/newshub/t7_counts.csv"
TS="$(date -Is)"

# psql が失敗しても空欄で埋めて継続
entity_linked=""
event_count=""
event_participant=""
ingested_events="-"
linked_entities="-"

  entity_linked=$(psql "$DATABASE_URL" -Atc "SELECT count(*) FROM entity WHERE ext_id IS NOT NULL;" 2>/dev/null || echo "")
  event_count=$(psql "$DATABASE_URL" -Atc "SELECT count(*) FROM event;" 2>/dev/null || echo "")
  event_participant=$(psql "$DATABASE_URL" -Atc "SELECT count(*) FROM event_participant;" 2>/dev/null || echo "")
fi

# 直近実行ログから件数を拾う（なければ "-"）

ie_line=$(journalctl -u events_ingest.service --since "-40 min" --no-pager | grep -E "ingested_events=" | tail -n1 || true)

fi
le_line=$(journalctl -u linking.service --since "-40 min" --no-pager | grep -E "linked_entities=" | tail -n1 || true)
if [[ -n "$le_line" ]]; then
  linked_entities=$(echo "$le_line" | sed -n 's/.*linked_entities=\([0-9]\+\).*/\1/p')
fi

# CSV ヘッダ
if [[ ! -s "$CSV" ]]; then
  echo "ts_iso,entity_linked,event_count,event_participant,ingested_events_last,linked_entities_last" | sudo tee -a "$CSV" >/dev/null
fi

# 追記
echo "$TS,${entity_linked:-},${event_count:-},${event_participant:-},${ingested_events:-},${linked_entities:-}" \
 | sudo tee -a "$CSV" >/dev/null
SH

# 3) 実行権限
sudo chmod 755 /usr/local/bin/newshub-t7-sampler.sh

# 4) まず1回だけ手動実行（エラー出ないか確認）
```
